Python Plotting Script

Use this script to load all 21 responses (or live exported CSV data) into a publication-ready ternary plot using mpltern in Python. 

# Python Program: Plot Roses & Champagne Value Responses
# Install dependencies: pip install matplotlib mpltern pandas

import matplotlib.pyplot as plt
import mpltern
import pandas as pd

# Load CSV generated from the website
df = pd.read_csv("ternary_responses.csv")

fig = plt.figure(figsize=(10, 8), dpi=150)
ax = fig.add_subplot(projection='ternary')

# Top = Equality, Left = Environmental Sustainability, Right = Human Dignity
ax.scatter(df['Present_Equality'], df['Present_Sustainability'], df['Present_Dignity'],
           color='#e8a598', alpha=0.7, s=60, edgecolors='#3d0c15', label='Present (2026)')

ax.scatter(df['Future_Equality'], df['Future_Sustainability'], df['Future_Dignity'],
           color='#f7e7ce', alpha=0.8, s=70, marker='^', edgecolors='#3d0c15', label='Future (2060)')

ax.set_tlabel("Equality")
ax.set_llabel("Environmental Sustainability")
ax.set_rlabel("Human Dignity")

ax.grid(True, linestyle='--', alpha=0.3)
ax.legend(loc='upper right')
plt.title("Roses & Champagne: Societal Value Transformations", fontsize=14, pad=20)
plt.tight_layout()
plt.savefig("roses_champagne_ternary.png", dpi=300)
plt.show()
